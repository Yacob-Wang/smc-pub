"navigation.indexes" in features -%} {%- for sub in item.children -%} {%- if sub.is_index and ns.target is none -%} {%- set ns.target = sub -%} {%- endif -%} {%- endfor -%} {%- endif -%} {%- if ns.target is none -%} {%- set first = item.children | first -%} {%- if first.children -%} {{- resolve_href(first) -}} {%- else -%} {{- first.url | url -}} {%- endif -%} {%- else -%} {{- ns.target.url | url -}} {%- endif -%} {%- endif -%} {%- endmacro %} {% macro resolve_tab_href(nav_item) -%} {#- 意图分组 Tab：可指定落点子项 title（见 mkdocs.yml extra.tab_landings） -#} {%- set landings = preferred and nav_item.children -%} {%- for child in nav_item.children -%} {%- if child.title == preferred and not ns.url -%} {%- set ns.url = resolve_href(child) -%} {%- endif -%} {%- endfor -%} {%- endif -%} {%- if ns.url -%} {{- ns.url -}} {%- elif nav_item.url -%} {{- nav_item.url | url -}} {%- else -%} {{- resolve_href(nav_item) -}} {%- endif -%} {%- endmacro %} {% macro
